% Odia Language Book XeLaTeX Preamble
% Configured for complex Indic script rendering and book styling

\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX}

% Spacing & paragraph flow
\usepackage{setspace}
\onehalfspacing
\setlength{\parindent}{0pt}
\setlength{\parskip}{7pt plus 2pt minus 1pt}

% Headers and Footers
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\color{gray}\leftmark}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% Unicode symbol mappings for arrow and bullet
\catcode`\•=\active
\def•{\ensuremath{\bullet}}
\catcode`\→=\active
\def→{\ensuremath{\rightarrow}}

% Start each chapter (top-level section) on a new page
\AddToHook{cmd/section/before}{\clearpage}

% Cover Page Background (Subtle Sambalpuri Ikat geometric vector pattern on Page 1)
\usepackage{eso-pic}
\usepackage{graphicx}

\IfFileExists{shared/assets/patterns/cover-bg.pdf}{%
  \AddToHook{begindocument/end}{%
    \AddToShipoutPictureBG*{%
      \AtPageLowerLeft{%
        \includegraphics[width=\paperwidth,height=\paperheight]{shared/assets/patterns/cover-bg.pdf}%
      }%
    }%
  }%
}{}

% Dedicated title/cover page: suppress headers/footers on page 1 and clearpage after title
\AddToHook{cmd/maketitle/after}{%
  \thispagestyle{empty}%
  \clearpage
}
